javascript:

(function () {

	'use strict';


	/* =====================================================
	   ROOT
	   ===================================================== */

	const root =
		fragmentElement.querySelector(
			'[data-pfl-blog-listing]'
		);


	if (!root) {
		return;
	}


	/* =====================================================
	   CONFIGURATION
	   ===================================================== */

	const OBJECT_ENDPOINT =
		'/o/c/pflblogslistings';


	/*
	 * Number of cards per request.
	 */

	const PAGE_SIZE = 6;


	/*
	 * Blog detail page.
	 *
	 * Example:
	 *
	 * /blog/my-blog-title
	 */

	const DETAIL_BASE_URL = '/blog/';


	/*
	 * IMPORTANT:
	 *
	 * Change these only if your Liferay
	 * Object field names are different.
	 */

	const FIELD_MAP = {

		title: 'title',

		category: 'category',

		image: 'image',

		excerpt: 'excerpt',

		authorName: 'authorName',

		authorImage: 'authorImage',

		publishDate: 'publishDate',

		readTime: 'readTime',

		slug: 'slug'

	};


	/* =====================================================
	   GET CURRENT SITE ID
	   ===================================================== */

	function getSiteId() {

		/*
		 * Preferred method.
		 */

		if (
			window.Liferay &&
			Liferay.ThemeDisplay &&
			typeof Liferay.ThemeDisplay
				.getSiteGroupId === 'function'
		) {

			return String(
				Liferay.ThemeDisplay
					.getSiteGroupId()
			);

		}


		/*
		 * Fallback.
		 */

		if (
			window.Liferay &&
			Liferay.ThemeDisplay &&
			typeof Liferay.ThemeDisplay
				.getScopeGroupId === 'function'
		) {

			return String(
				Liferay.ThemeDisplay
					.getScopeGroupId()
			);

		}


		return '';

	}


	const SITE_ID =
		getSiteId();


	/*
	 * Final site-scoped Object endpoint.
	 *
	 * Example:
	 *
	 * /o/c/pflblogslistings/scopes/20123
	 */

	const API_URL =
		OBJECT_ENDPOINT +
		'/scopes/' +
		encodeURIComponent(
			SITE_ID
		);


	/* =====================================================
	   DOM
	   ===================================================== */

	const tabs =
		root.querySelector(
			'[data-blog-tabs]'
		);


	const grid =
		root.querySelector(
			'[data-blog-grid]'
		);


	const loading =
		root.querySelector(
			'[data-blog-loading]'
		);


	const empty =
		root.querySelector(
			'[data-blog-empty]'
		);


	const error =
		root.querySelector(
			'[data-blog-error]'
		);


	const sort =
		root.querySelector(
			'[data-blog-sort]'
		);


	const loadMoreWrapper =
		root.querySelector(
			'[data-load-more-wrapper]'
		);


	const loadMoreButton =
		root.querySelector(
			'[data-load-more]'
		);


	/* =====================================================
	   STATE
	   ===================================================== */

	let activeCategory = 'ALL';

	let activeCategoryKey = '';

	let currentSort = 'latest';

	let currentPage = 1;

	let totalCount = 0;

	let loadedCount = 0;

	let loadingRequest = false;


	/* =====================================================
	   START
	   ===================================================== */

	init();


	async function init() {

		if (!SITE_ID) {

			showError(
				'Unable to determine the current site.'
			);

			return;

		}


		bindEvents();


		await loadCategoriesAndInitialBlogs();

	}


	/* =====================================================
	   EVENTS
	   ===================================================== */

	function bindEvents() {

		/*
		 * Sort
		 */

		if (sort) {

			sort.addEventListener(
				'change',
				function () {

					currentSort =
						this.value;

					currentPage = 1;

					loadedCount = 0;

					loadBlogs(true);

				}
			);

		}


		/*
		 * Load more
		 */

		if (loadMoreButton) {

			loadMoreButton.addEventListener(
				'click',
				function () {

					if (
						loadingRequest
					) {

						return;

					}


					currentPage++;


					loadBlogs(false);

				}
			);

		}

	}


	/* =====================================================
	   INITIAL LOAD
	   ===================================================== */

	async function loadCategoriesAndInitialBlogs() {

		showLoading();

		hideError();

		hideEmpty();


		try {

			/*
			 * Fetch enough records to discover
			 * the categories.
			 *
			 * We do not use this response
			 * for the final grid pagination.
			 */

			const params =
				new URLSearchParams();


			params.set(
				'page',
				'1'
			);


			params.set(
				'pageSize',
				'100'
			);


			params.set(
				'sort',
				getSortExpression()
			);


			const response =
				await fetch(
					API_URL +
					'?' +
					params.toString(),
					{
						method: 'GET',

						headers: {
							'Accept':
								'application/json'
						},

						credentials: 'same-origin'
					}
				);


			if (!response.ok) {

				throw new Error(
					'HTTP ' +
					response.status
				);

			}


			const data =
				await response.json();


			const items =
				Array.isArray(
					data.items
				)
					? data.items
					: [];


			/*
			 * Build category tabs.
			 */

			buildCategoryTabs(
				items
			);


			/*
			 * Instead of rendering all 100,
			 * render only PAGE_SIZE.
			 */

			const initialItems =
				items.slice(
					0,
					PAGE_SIZE
				);


			totalCount =
				data.totalCount ||
				items.length;


			loadedCount =
				initialItems.length;


			renderBlogs(
				initialItems,
				true
			);


			updateLoadMore();

		}
		catch (e) {

			console.error(
				'PFL BLOG API ERROR:',
				e
			);


			showError(
				'Unable to load blog posts.'
			);

		}
		finally {

			hideLoading();

		}

	}


	/* =====================================================
	   LOAD BLOGS
	   ===================================================== */

	async function loadBlogs(
		reset
	) {

		if (loadingRequest) {
			return;
		}


		loadingRequest = true;


		hideError();


		if (reset) {

			showLoading();

			hideEmpty();

			grid.innerHTML = '';

		}
		else {

			loadMoreButton.disabled =
				true;

			loadMoreButton.textContent =
				'Loading...';

		}


		try {

			const params =
				new URLSearchParams();


			params.set(
				'page',
				String(
					currentPage
				)
			);


			params.set(
				'pageSize',
				String(
					PAGE_SIZE
				)
			);


			params.set(
				'sort',
				getSortExpression()
			);


			/*
			 * CATEGORY FILTER
			 *
			 * For a Text field:
			 *
			 * category eq 'Credit Score'
			 *
			 * For a Picklist:
			 *
			 * category eq 'creditScore'
			 *
			 * We use the Picklist key below.
			 */

			if (
				activeCategory !== 'ALL' &&
				activeCategoryKey
			) {

				const filterValue =
					activeCategoryKey
						.replace(
							/'/g,
							"''"
						);


				params.set(
					'filter',
					FIELD_MAP.category +
					" eq '" +
					filterValue +
					"'"
				);

			}


			console.log(
				'PFL Blog API:',
				API_URL +
				'?' +
				params.toString()
			);


			const response =
				await fetch(
					API_URL +
					'?' +
					params.toString(),
					{
						method: 'GET',

						headers: {
							'Accept':
								'application/json'
						},

						credentials: 'same-origin'
					}
				);


			if (!response.ok) {

				throw new Error(
					'HTTP ' +
					response.status
				);

			}


			const data =
				await response.json();


			const items =
				Array.isArray(
					data.items
				)
					? data.items
					: [];


			totalCount =
				data.totalCount ||
				items.length;


			if (reset) {

				loadedCount =
					items.length;

			}
			else {

				loadedCount +=
					items.length;

			}


			renderBlogs(
				items,
				reset
			);


			/*
			 * If the filtered response
			 * has no entries.
			 */

			if (
				reset &&
				items.length === 0
			) {

				showEmpty();

			}


			updateLoadMore();

		}
		catch (e) {

			console.error(
				'PFL BLOG API ERROR:',
				e
			);


			showError(
				'Unable to load blog posts.'
			);

		}
		finally {

			loadingRequest = false;

			hideLoading();


			if (loadMoreButton) {

				loadMoreButton.disabled =
					false;

				loadMoreButton.textContent =
					'Load more';

			}

		}

	}


	/* =====================================================
	   BUILD CATEGORY TABS
	   ===================================================== */

	function buildCategoryTabs(
		items
	) {

		const categories = [];


		/*
		 * Always show ALL.
		 */

		categories.push({

			key: 'ALL',

			name: 'ALL'

		});


		items.forEach(
			function (item) {

				const category =
					getCategoryObject(
						item
					);


				if (!category) {
					return;
				}


				/*
				 * Avoid duplicate categories.
				 */

				const exists =
					categories.some(
						function (existing) {

							return (
								existing.key ===
								category.key
							);

						}
					);


				if (!exists) {

					categories.push(
						category
					);

				}

			}
		);


		/*
		 * Render.
		 */

		tabs.innerHTML =
			categories
				.map(
					function (category) {

						const isActive =
							category.key ===
							'ALL';


						return `
							<button
								type="button"
								class="pfl-blog-tab ${
									isActive
										? 'is-active'
										: ''
								}"
								data-category-key="${escapeHtml(
									category.key
								)}"
								data-category-name="${escapeHtml(
									category.name
								)}"
								role="tab"
								aria-selected="${
									isActive
								}"
							>
								${escapeHtml(
									category.name
								)}
							</button>
						`;

					}
				)
				.join('');


		/*
		 * Bind category events.
		 */

		tabs
			.querySelectorAll(
				'.pfl-blog-tab'
			)
			.forEach(
				function (button) {

					button.addEventListener(
						'click',
						onCategoryClick
					);

				}
			);

	}


	/* =====================================================
	   CATEGORY CLICK
	   ===================================================== */

	function onCategoryClick(
		event
	) {

		const button =
			event.currentTarget;


		activeCategory =
			button.dataset.categoryName ||
			'ALL';


		activeCategoryKey =
			button.dataset.categoryKey ||
			'';


		/*
		 * ALL
		 */

		if (
			activeCategoryKey ===
			'ALL'
		) {

			activeCategory =
				'ALL';

			activeCategoryKey =
				'';

		}


		/*
		 * Active class.
		 */

		tabs
			.querySelectorAll(
				'.pfl-blog-tab'
			)
			.forEach(
				function (tab) {

					const isActive =
						tab === button;


					tab.classList.toggle(
						'is-active',
						isActive
					);


					tab.setAttribute(
						'aria-selected',
						String(
							isActive
						)
					);

				}
			);


		/*
		 * Reset page.
		 */

		currentPage = 1;

		loadedCount = 0;


		/*
		 * Load.
		 */

		loadBlogs(true);

	}


	/* =====================================================
	   GET CATEGORY OBJECT
	   ===================================================== */

	function getCategoryObject(
		item
	) {

		if (!item) {
			return null;
		}


		const value =
			item[
				FIELD_MAP.category
			];


		if (!value) {
			return null;
		}


		/*
		 * Liferay Picklist:
		 *
		 * {
		 *     "key": "creditScore",
		 *     "name": "Credit Score"
		 * }
		 */

		if (
			typeof value ===
			'object' &&
			!Array.isArray(value)
		) {

			if (
				value.key
			) {

				return {

					key:
						String(
							value.key
						),

					name:
						String(
							value.name ||
							value.key
						)

				};

			}

			if (
				value.name
			) {

				return {

					key:
						String(
							value.name
						),

					name:
						String(
							value.name
						)

				};

			}

		}


		/*
		 * Plain text category.
		 */

		if (
			typeof value ===
			'string'
		) {

			return {

				key: value,

				name: value

			};

		}


		return null;

	}


	/* =====================================================
	   RENDER BLOGS
	   ===================================================== */

	function renderBlogs(
		items,
		reset
	) {

		if (reset) {

			grid.innerHTML = '';

		}


		if (!items.length) {

			if (reset) {

				showEmpty();

			}

			return;

		}


		hideEmpty();


		const html =
			items
				.map(
					renderBlogCard
				)
				.join('');


		grid.insertAdjacentHTML(
			'beforeend',
			html
		);

	}


	/* =====================================================
	   BLOG CARD
	   ===================================================== */

	function renderBlogCard(
		item
	) {

		const title =
			getValue(
				item,
				FIELD_MAP.title
			) ||
			'Untitled Blog';


		const category =
			getCategoryObject(
				item
			);


		const categoryName =
			category
				? category.name
				: '';


		const excerpt =
			getValue(
				item,
				FIELD_MAP.excerpt
			) ||
			'';


		const authorName =
			getValue(
				item,
				FIELD_MAP.authorName
			) ||
			'';


		const publishDate =
			getValue(
				item,
				FIELD_MAP.publishDate
			);


		const readTime =
			getValue(
				item,
				FIELD_MAP.readTime
			);


		const imageValue =
			getValue(
				item,
				FIELD_MAP.image
			);


		const authorImageValue =
			getValue(
				item,
				FIELD_MAP.authorImage
			);


		const imageUrl =
			getImageUrl(
				imageValue
			);


		const authorImageUrl =
			getImageUrl(
				authorImageValue
			);


		const slug =
			getValue(
				item,
				FIELD_MAP.slug
			);


		const detailUrl =
			getDetailUrl(
				item,
				slug
			);


		return `

			<article
				class="pfl-blog-card"
			>

				<a
					class="pfl-blog-card-link"
					href="${escapeHtml(
						detailUrl
					)}"

					>

							categoryName
								?
								`
								<div
									class="pfl-blog-card-category"
								>
									${escapeHtml(
										categoryName
									)}
								</div>
								`
								:
								''
						}


						<h2
							class="pfl-blog-card-title"
						>
							${escapeHtml(
								title
							)}
						</h2>


						${
							excerpt
								?
								`
								<p
									class="pfl-blog-card-excerpt"
								>
									${escapeHtml(
										stripHtml(
											excerpt
										)
									)}
								</p>
								`
								:
								''
						}


						${
							authorName ||
							publishDate ||
							readTime
								?
								`
								<div
									class="pfl-blog-card-meta"

												src="${escapeHtml(

												)}"

												loading="lazy"

									}

									>

												>

	/* =====================================================
	   GET VALUE
	   ===================================================== */

	function getValue(
		object,
		field
	) {

		if (
			!object ||
			!field
		) {

			return '';

		}


		return object[field];

	}


	/* =====================================================
	   IMAGE URL
	   ===================================================== */

	function getImageUrl(
		value
	) {

		if (!value) {
			return '';
		}


		/*
		 * String URL.
		 */

		if (
			typeof value ===
			'string'
		) {

			return value;

		}


		/*
		 * Liferay attachment.
		 */

		if (
			value.contentUrl
		) {

			return value.contentUrl;

		}


		/*
		 * URL property.
		 */

		if (
			value.url
		) {

			return value.url;

		}


		/*
		 * Link object.
		 */

		if (
			value.link
		) {

			if (
				typeof value.link ===
				'string'
			) {

				return value.link;

			}


			if (
				value.link.href
			) {

				return value.link.href;

			}

		}


		return '';

	}


	/* =====================================================
	   DETAIL URL
	   ===================================================== */

	function getDetailUrl(
		item,
		slug
	) {

		/*
		 * If your Object has a direct
		 * URL field, use it.
		 */

		if (
			item.viewUrl
		) {

			return item.viewUrl;

		}


		if (
			item.url
		) {

			return item.url;

		}


		/*
		 * Otherwise use slug.
		 */

		if (slug) {

			return (
				DETAIL_BASE_URL +
				encodeURIComponent(
					String(
						slug
					)
				)
			);

		}


		/*
		 * Fallback.
		 */

		return '#';

	}


	/* =====================================================
	   SORT
	   ===================================================== */

	function getSortExpression() {

		if (
			currentSort ===
			'oldest'
		) {

			return (
				FIELD_MAP.publishDate +
				':asc'
			);

		}


		return (
			FIELD_MAP.publishDate +
			':desc'
		);

	}


	/* =====================================================
	   DATE FORMAT
	   ===================================================== */

	function formatDate(
		value
	) {

		if (!value) {
			return '';
		}


		try {

			const date =
				new Date(
					value
				);


			if (
				isNaN(
					date.getTime()
				)
			) {

				return String(
					value
				);

			}


			return new Intl.DateTimeFormat(
				'en-GB',
				{
					day: 'numeric',

					month: 'short',

					year: 'numeric'
				}
			).format(
				date
			);

		}
		catch (e) {

			return String(
				value
			);

		}

	}


	/* =====================================================
	   READ TIME
	   ===================================================== */

	function formatReadTime(
		value
	) {

		if (!value) {
			return '';
		}


		if (
			typeof value ===
			'number'
		) {

			return (
				value +
				' Minutes'
			);

		}


		return String(
			value
		);

	}


	/* =====================================================
	   LOAD MORE
	   ===================================================== */

	function updateLoadMore() {

		if (!loadMoreWrapper) {
			return;
		}


		/*
		 * No more entries.
		 */

		if (
			loadedCount >=
			totalCount
		) {

			loadMoreWrapper
				.classList
				.add(
					'is-hidden'
				);

			return;

		}


		/*
		 * More entries exist.
		 */

		loadMoreWrapper
			.classList
			.remove(
				'is-hidden'
			);

	}

	   LOADING

			loading.classList.remove(

			);

		}

	}

		if (loading) {

			loading.classList.add(

			);

		}

	}

	function showEmpty() {

		if (empty) {

			empty.classList.add(
				'is-visible'
			);

		}

	}

		if (empty) {

			empty.classList.remove(

			);

		}

	}

	   ERROR
	   ===================================================== */

	function showError(
		message
	) {

		hideLoading();

		hideEmpty();


		if (error) {

			error.textContent =
				message ||
				'Unable to load blog posts.';

			error.classList.add(
				'is-visible'
			);

		}

	}


	function hideError() {

		if (error) {

			error.classList.remove(
				'is-visible'
			);

		}

	}


	/* =====================================================
	   STRIP HTML
	   ===================================================== */

	function stripHtml(
		value
	) {

		if (!value) {
			return '';
		}


		const temporary =
			document.createElement(
				'div'
			);


		temporary.innerHTML =
			String(value);


		return (
			temporary.textContent ||
			temporary.innerText ||
			''
		);

	}


	/* =====================================================
	   ESCAPE HTML
	   ===================================================== */

	function escapeHtml(
		value
	) {

		return String(
			value ?? ''
		)
			.replace(
				/&/g,
				'&amp;'
			)
			.replace(
				/</g,
				'&lt;'
			)
			.replace(
				/>/g,
				'&gt;'
			)
			.replace(
				/"/g,
				'&quot;'
			)
			.replace(
				/'/g,
				'&#039;'
			);

	}
